"""收尾只读审计：原始文件哈希、源码快照、JSON格式与新文档链接；不重写实验记录。"""
import hashlib
import json
from pathlib import Path
import re


base = Path('docs/results/2026-10-26')
audit = dict(rawFilesVerified=0,jsonValuesParsed=0,linksVerified=0,sourceFilesVerified=0)
for name in ('t6-tool-100-01','t6-formal-100-01','t6-scale-500-01'):
    folder = base / name
    manifest = json.loads((folder / 'raw-sha256.json').read_text())
    for entry in manifest:
        path = folder / entry['path']
        assert hashlib.sha256(path.read_bytes()).hexdigest() == entry['sha256'], str(path)
        audit['rawFilesVerified'] += 1
    # 只验证自身实验新目录，不触碰历史T2/T3/T4/T5的manifest。
    for path in folder.rglob('*.json'):
        json.loads(path.read_text(encoding='utf-8-sig'))
        audit['jsonValuesParsed'] += 1
    for path in folder.rglob('*.jsonl'):
        for line in path.read_text(encoding='utf-8-sig').splitlines():
            json.loads(line)
            audit['jsonValuesParsed'] += 1
    for entry in json.loads((folder / 'source-sha256.json').read_text(encoding='utf-8-sig')):
        assert hashlib.sha256(Path(entry['path']).read_bytes()).hexdigest().upper() == entry['sha256']
        audit['sourceFilesVerified'] += 1
snapshots = {'scripts/load/reconnect.py':'reconnect.py.txt','scripts/run/reconnect.ps1':'reconnect.ps1.txt',
    'scripts/tests/test_reconnect.py':'test-reconnect.py.txt','scripts/tests/reconnect-summary.test.py':'verify-summary.py.txt',
    'scripts/load/longrun.py':'longrun.py.txt','scripts/run/ProcessEvidence.psm1':'ProcessEvidence.psm1.txt'}
for source, target in snapshots.items():
    assert Path(source).read_bytes() == (base / 't6-verification' / target).read_bytes()
docs = [Path('docs/RECONNECT.md'),base/'t6-report.md',base/'t6-verification/README.md',
        base/'t6-formal-100-01/notes.md',base/'t6-scale-500-01/notes.md']
for doc in docs:
    for target in re.findall(r'\]\(([^)]+)\)', doc.read_text(encoding='utf-8')):
        assert (doc.parent / target.split('#')[0]).exists(), (str(doc),target)
        audit['linksVerified'] += 1
audit['snapshotFilesVerified'] = len(snapshots)
print(json.dumps(audit,indent=2))
(base/'t6-verification/additional-audit.json').write_text(json.dumps(audit,indent=2)+'\n',encoding='utf-8')
