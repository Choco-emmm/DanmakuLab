"""长测的轻量只读进度：正常接收是客户端回调计数，最终仍需逐观众结算。"""
from pathlib import Path
import json

out=Path(__file__).resolve().parent
read=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
state=read(out/'batch-status.json');raw=Path(state['rawRoot'])
current=max((p for p in raw.iterdir() if p.is_dir()),key=lambda p:p.stat().st_ctime)
def tail(path):
    if not path.exists():return None
    with path.open('rb') as stream:
        stream.seek(max(0,path.stat().st_size-65536));lines=stream.read().splitlines()
    for line in reversed(lines):
        try:return json.loads(line)
        except (ValueError,UnicodeError):pass
    return None
client=tail(current/'load/client/progress.jsonl')
sample=tail(current/'load/samples.jsonl')
metrics={m['name']:m['measurements'][0]['value'] for m in (sample or {}).get('snapshot',{}).get('metrics',[])}
print(json.dumps(dict(case=current.name,formalPassed=state['formalPassed'],client=client,
    connections=metrics.get('lab.connections'),outputDrops=metrics.get('lab.outbound.dropped'),
    writesFailed=metrics.get('lab.writes.failed'),collectorError=(sample or {}).get('error'),
    finalError=state['error']),ensure_ascii=False))
