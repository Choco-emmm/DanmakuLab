"""轻量只读进度：服务器提交量不代表真实接收，终态仍由独立审计决定。"""
from pathlib import Path
import json,time

out=Path(__file__).resolve().parent
read=lambda p:json.loads(p.read_text(encoding='utf-8-sig'))
state=read(out/'batch-status.json')
raw=Path(state['rawRoot'])
configs=[(read(p),p.parent) for p in raw.glob('*/config.json')]
cfg,current=max(configs,key=lambda item:item[0]['startAtMs'])
elapsed=(time.time()*1000-cfg['startAtMs'])/1000
phase='preparation' if elapsed<0 else ('normal-only','hot-20','hot-100','hot-200')[int(elapsed//cfg['phaseSeconds'])] if elapsed<4*cfg['phaseSeconds'] else 'drain/closing'
sample=None
path=current/'samples.jsonl'
if path.exists():
    with path.open('rb') as f:
        f.seek(max(0,path.stat().st_size-65536));lines=f.read().splitlines()
    for line in reversed(lines):
        try:sample=json.loads(line);break
        except (ValueError,UnicodeError):pass
metrics={m['name']:m['measurements'][0]['value'] for m in (sample or {}).get('snapshot',{}).get('metrics',[])}
selected={k:metrics.get(k) for k in ('lab.connections','lab.messages.input','lab.messages.accepted','lab.messages.rejected',
    'lab.writes.failed','lab.outbound.dropped','lab.outbound.messages','lab.dispatch.messages','lab.outbound.writes.inflight')}
print(json.dumps(dict(case=current.name,phase=phase,secondsFromInputStart=round(elapsed),formalPassed=state['formalCompleted'],
    metrics=selected,collectorError=(sample or {}).get('error'),lastResult=state['rows'][-1]['name'] if state['rows'] else None),ensure_ascii=False))
