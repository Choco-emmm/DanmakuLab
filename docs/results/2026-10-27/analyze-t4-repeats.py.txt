"""从独立校验后的T4重复结果生成派生统计和图；不修改任何原始样本或manifest。

用法：python -X utf8 docs/results/2026-10-27/analyze-t4-repeats.py.txt 批次目录
图中每个点保留某轮、某模式的原房间P99，不将多轮分位数平均成“整体P99”。
"""
import argparse
import json
import math
from pathlib import Path
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt


def read(path):
    return json.loads(path.read_text(encoding='utf-8-sig'))


def resources(folder):
    """进程CPU以相邻累计CPU秒/采样墙钟秒计算等效逻辑核；峰值仅指采样点最大值。"""
    rows = [json.loads(line) for line in (folder / 'samples.jsonl').read_text(encoding='utf-8-sig').splitlines()]
    assert rows and all('error' not in row for row in rows)
    cpu = {role: [] for role in ('java', 'normal', 'hot')}
    heap, working_set = [], []
    maxima = {}
    for index, row in enumerate(rows):
        heaps = [item for item in row['snapshot']['metrics'] if item['name'] == 'jvm.memory.used']
        heap.append(heaps[-1]['measurements'][0]['value'])
        working_set.append(row['server']['workingSetBytes'])
        # 各Gauge是顺序采样；highwater由服务端维护，比单看五秒采样更能发现保留预算的峰值。
        for item in row['snapshot']['metrics']:
            if item['name'].startswith('lab.'):
                value = item['measurements'][0]['value']
                maxima[item['name']] = max(maxima.get(item['name'], 0), value)
        if index:
            before = rows[index - 1]
            seconds = row['elapsedSeconds'] - before['elapsedSeconds']
            for role in cpu:
                current = row['server'] if role == 'java' else row['clients'].get(role)
                previous = before['server'] if role == 'java' else before['clients'].get(role)
                if current and previous and current['pid'] == previous['pid'] and seconds > 0:
                    delta = current['cpuSeconds'] - previous['cpuSeconds']
                    assert delta >= 0
                    cpu[role].append((delta, seconds))
    # 最终快照的服务端高水位覆盖最后一个五秒采样点之后的收尾；堆/工作集仍只报告采样最大。
    after = read(folder / 'server-after.json')
    for item in after['metrics']:
        if item['name'].endswith('.highwater'):
            maxima[item['name']] = max(maxima.get(item['name'], 0), item['measurements'][0]['value'])
    assert maxima['lab.connections'] <= 1000
    assert maxima['lab.outbound.bytes.highwater'] <= 134217728
    assert maxima.get('lab.dispatch.bytes.highwater', 0) <= 4194304
    return dict(samples=len(rows), heapSampledMaxMiB=max(heap) / 1048576,
                javaWorkingSetSampledMaxMiB=max(working_set) / 1048576,
                retainedPayloadHighWaterBytes=dict(outbound=maxima['lab.outbound.bytes.highwater'],
                                                   dispatch=maxima.get('lab.dispatch.bytes.highwater', 0)),
                sampledMaxInFlightWrites=maxima['lab.outbound.writes.inflight'],
                meanEquivalentCores={role: sum(x for x, t in values) / sum(t for x, t in values) if values else None
                                     for role, values in cpu.items()})


def analyze(folder):
    summary = read(folder / 'verified-summary.json')
    assert summary['passed'] and summary['formal']
    resource_rows = []
    table = []
    names = ('normal only', 'hot input 20/s', 'hot input 100/s', 'hot input 200/s')
    fig, axes = plt.subplots(2, 1, figsize=(10, 8), sharex=True)
    colors = {'direct': '#c96a24', 'fair': '#176b99'}
    styles = ('-', '--', ':')
    for row in summary['runs']:
        normal_worst = [max(phase['normalP99ByRoom']) for phase in row['phases']]
        hot_p99 = [phase['hotP99Ms'] if phase['hotP99Ms'] is not None else math.nan for phase in row['phases']]
        label = f"{row['mode']} run {row['round']}"
        for ax, values in zip(axes, (normal_worst, hot_p99)):
            ax.plot(range(4), values, color=colors[row['mode']], linestyle=styles[(row['round'] - 1) % 3],
                    marker='o' if row['mode'] == 'direct' else 's', label=label, linewidth=1.8)
        resource_rows.append(dict(round=row['round'], mode=row['mode'],
                                 **resources(folder / row['directory'] / 'load')))
        table.append(dict(round=row['round'], mode=row['mode'], received=row['actual']['received'],
            accepted=row['actual']['accepted'], rejected=row['actual']['rejected'],
            ordinaryP99RangesMs=[[min(p['normalP99ByRoom']), max(p['normalP99ByRoom'])] for p in row['phases']],
            ordinaryP95MaxMs=[max(p['normalP95ByRoom']) for p in row['phases']],
            hotP99Ms=[p['hotP99Ms'] for p in row['phases']],
            hotInputPerSecond=[p['hotInputPerSecond'] for p in row['phases']],
            hotMaxSendLagMs=[p['hotMaxLagMs'] for p in row['phases']]))
    axes[0].set_title('Ordinary rooms: worst individual-room P99 in each run')
    axes[1].set_title('Hot room: send-to-callback P99, including client receive processing')
    for ax in axes:
        ax.set_ylabel('P99 (ms)')
        ax.set_ylim(bottom=0)
        ax.grid(alpha=.2)
    axes[0].legend(ncol=3, fontsize=9)
    axes[1].set_xticks(range(4), names)
    fig.suptitle('T4 independent repetitions | same JAR | 1,000 viewers | 180 seconds per phase', fontsize=12)
    fig.tight_layout()
    fig.savefig(folder / 'latency-comparison.png', dpi=170)
    fig.savefig(folder / 'latency-comparison.svg')
    plt.close(fig)
    report = dict(jarSha256=summary['jarSha256'], actualReceipts=summary['received'], runs=table, resources=resource_rows,
                  note='Same-host sequential runs; percentile ranges preserve individual room/run values. Resource maxima are sampled.')
    (folder / 'analysis.json').write_text(json.dumps(report, indent=2) + '\n', encoding='utf-8')
    print(json.dumps(report, indent=2))


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('directory', type=Path)
    analyze(parser.parse_args().directory)
